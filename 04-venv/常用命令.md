```python
sudo apt update
sudo apt install python3-venv
python3 -m venv .venv       # python3 -m venv xxx . 在 Linux 中表示隐藏目录，这样不会让虚拟环境目录占据项目目录的视觉空间


#windows
.venv\Scripts\activate #激活
rmdir /s /q .venv	#删除
where python

#linux
source .venv/bin/activate #激活
rm -rf .venv				#删除
which python

deactivate #退出

python --version

pip install requests==2.32.3 #安装包  == 可以指定版本

pip list 			#查看已安装的包
pip show requests  #查看某个包

pip freeze > requirements.txt  #导出依赖
pip install -r requirements.txt
```
